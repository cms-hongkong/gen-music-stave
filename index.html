async function exportImage() { const svgs = document.querySelectorAll(".vexflow-page svg"); if (svgs.length === 0) { alert("沒有樂譜可以匯出！"); return; } const titleEl = document.getElementById('displayTitle'); const titleText = titleEl && !titleEl.classList.contains('hidden') ? titleEl.innerText : ""; const baseName = titleText ? titleText : "Clare_Music_Score"; const exportScale = 2; // 2 倍高畫質匯出 // 1. 計算所有頁面的總高度與最大寬度 let totalHeight = (titleText ? 80 : 20); let maxWidth = 0; let pageHeights = []; for (let i = 0; i < svgs.length; i++) { const svg = svgs[i]; const width = parseInt(svg.getAttribute("width")) || svg.getBoundingClientRect().width; const height = parseInt(svg.getAttribute("height")) || svg.getBoundingClientRect().height; if (width > maxWidth) maxWidth = width; pageHeights.push(height); totalHeight += height + 20; // 每頁底部加入 20px 緩衝區 } // 2. 建立單一超大畫布，用來容納所有樂譜 const canvas = document.createElement("canvas"); canvas.width = (maxWidth + 40) * exportScale; canvas.height = totalHeight * exportScale; const ctx = canvas.getContext("2d"); ctx.scale(exportScale, exportScale); // 填滿純白背景（避免背景變透明） ctx.fillStyle = "white"; ctx.fillRect(0, 0, canvas.width, canvas.height); // 繪製大標題 if (titleText) { ctx.fillStyle = "#0f172a"; ctx.font = "bold 34px sans-serif"; ctx.textAlign = "center"; ctx.fillText(titleText, (maxWidth + 40) / 2, 50); } let currentY = titleText ? 80 : 20; // 3. 逐頁安全地將 SVG 繪製入畫布中 for (let i = 0; i < svgs.length; i++) { const svg = svgs[i]; let svgString = new XMLSerializer().serializeToString(svg); if (!svgString.includes('xmlns="http://www.w3.org/2000/svg"')) { svgString = svgString.replace(' { const img = new Image(); img.onload = function () { ctx.drawImage(img, 20, currentY); URL.revokeObjectURL(url); // 清理記憶體 resolve(); }; img.onerror = function() { URL.revokeObjectURL(url); resolve(); }; img.src = url; }); currentY += pageHeights[i] + 20; } // 4. 觸發單次圖片下載（合併成一張長圖） try { const link = document.createElement("a"); link.download = baseName + ".png"; link.href = canvas.toDataURL("image/png", 1.0); document.body.appendChild(link); link.click(); document.body.removeChild(link); } catch (e) { alert("圖片產生失敗，可能是樂譜過長導致畫布超載。這種情況請使用「匯出 PDF」。"); } }
